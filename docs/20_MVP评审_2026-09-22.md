### 7.2 决定性证据（可复现）

探针与本仓集成测试同款——**真 ASGI 应用、两个租户、不打桩任何鉴权分支**；租户 A 有一条 `review` 态成品 `victim-1`，所有调用都用租户 B 的身份发起。

"""Q199 evidence probe: can tenant B adjudicate tenant A's content?

Run against the real ASGI app (same harness style as the repo's integration tests),
with two tenants and no monkeypatching of authorization. This turns the #32
finding from code-reading into executed evidence.
"""

import asyncio

from httpx import ASGITransport, AsyncClient
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
from sqlalchemy.pool import StaticPool

from app.content.models import CONTENT_READY, ContentProduct
from app.core.db import Base, get_session
from app.main import app

ACTOR_B = {"id": "cust-B", "roles": []}


async def main() -> None:
    engine = create_async_engine(
        "sqlite+aiosqlite:///:memory:",
        connect_args={"check_same_thread": False},
        poolclass=StaticPool,
    )
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    factory = async_sessionmaker(engine, expire_on_commit=False, class_=AsyncSession)

    async def get_test_session():
        async with factory() as session:
            yield session

    app.dependency_overrides[get_session] = get_test_session
    async with factory() as session:
        session.add(
            ContentProduct(
                content_id="victim-1",
                tenant_id="tenant-A",
                product_space_id="ps-A",
                final_id="fcw-A",
                goal="种草",
                platform="douyin",
                status="review",
            )
        )
        await session.commit()

    async with AsyncClient(transport=ASGITransport(app=app), base_url="http://t") as c:
        read = await c.get("/api/content/victim-1", params={"tenant_id": "tenant-B"})
        print(f"读：tenant-B 取 tenant-A 的成品 → HTTP {read.status_code}"
              f" | 返回体 tenant_id={read.json().get('tenant_id') if read.status_code == 200 else '-'}")

        approve = await c.post(
            "/api/content/victim-1/approve", json={"actor": ACTOR_B}
        )
        print(f"写：tenant-B 裁决（approve）tenant-A 的成品 → HTTP {approve.status_code}")

        reject = await c.post(
            "/api/content/victim-1/reject", json={"actor": ACTOR_B, "reason": "串台"}
        )
        print(f"写：tenant-B 驳回 tenant-A 的成品 → HTTP {reject.status_code}")

        after = await c.get("/api/content/victim-1", params={"tenant_id": "tenant-A"})
        print(f"后果：tenant-A 侧看该成品 status → "
              f"{after.json().get('status') if after.status_code == 200 else after.status_code}")

        lang = await c.patch(
            "/api/intakes/does-not-exist/target-languages",
            json={"languages": ["zh-CN"], "actor": ACTOR_B},
        )
        print(f"旁证：intake 语言口未知 id → HTTP {lang.status_code}（该口同样不按租户定位）")

    app.dependency_overrides.clear()
